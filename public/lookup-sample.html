JavaScript Integration Code: Lookup & Upsert by UDISE 
// 1. GET: Fetch existing school info to pre-fill form
async function fetchSchoolByUDISE(udiseCode) {
  const res = await fetch(`/api/schools/udise/${encodeURIComponent(udiseCode)}`);
  if (res.status === 404) return null; // School not registered yet
  return await res.json();
}

// 2. POST: Upsert (Add new school or Update existing school with latest timestamp)
async function syncSchoolToSMDb(udiseCode, formData) {
  const res = await fetch(`/api/schools/udise/${encodeURIComponent(udiseCode)}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(formData)
  });
  
  const result = await res.json();
  console.log('Action performed:', result.action); // "updated" or "created"
  console.log('New Timestamp:', result.timestamp);
  return result;
}
